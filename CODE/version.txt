# The identity strings compiled into the Windows executables as a
# VERSIONINFO resource.  CODE/mkverres.bat reads this file and writes the
# .rc; CODE/gfortran_O3.bat and CODE/gfortran_O3_OMP.bat link the result.
#
# VERSION is the package version, and this is the only place it is written
# down.  Users run released versions and need to tell which one they have,
# so every release bumps it (and COMMA_VERSION with it), gets an entry in
# DOCS/CHANGES.md and a git tag v<VERSION>, and the four Windows
# executables are rebuilt so that the version they carry is the release's.
# A user reads it in Explorer, Properties > Details of the .exe, or in
# PowerShell with  (Get-Item RCE20KD.exe).VersionInfo.FileVersion .
# (The resource also has a second purpose: a PE with no resource directory
# at all is what Norton's Win64:Evo-gen heuristic reacts to.)
#
# COMMA_VERSION is the same number in the form the binary FILEVERSION
# field requires: four 16-bit fields.  Keeping both spellings here avoids
# doing arithmetic in a .bat, which is where this kind of thing goes
# wrong.  Keep the two in step by hand.
#
# ELF has no counterpart, so CODE/unix/build.sh reads nothing from here.
#
# COPYRIGHT is the one string here that has to stay true: it is compiled
# into every distributed binary and it used to read "Public domain", which
# contradicted LICENSE.md.  Keep it in step with LICENSE.md, and keep it
# free of ( ) < > & | ^ % -- mkverres.bat echoes it into the .rc and cmd
# would eat those.
VERSION=26.2.2
COMMA_VERSION=26,2,2,0
COMPANY=A. Kramida
PRODUCT=Cowan Atomic Structure Codes
COPYRIGHT=Copyright 2022-2026 A. Kramida. Original codes by R. D. Cowan. Free to distribute, not to sell -- see LICENSE.md.
